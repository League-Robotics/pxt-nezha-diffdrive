"""Sprint 024 ticket 001 bench check, on tovez over USB.

Deliberately RAW pyserial, not tools/robotlink.py: ticket 002 made
open_link() send HELLO on connect, which would put a request on the wire
and contaminate the silence measurement this script exists to make.

Phases:
  A  open port, send NOTHING, listen 3.0 s. Opening the port resets the
     target program (CLAUDE.md), so the one-time boot banner IS expected
     here -- sprint 024 leaves it in scope-out. What must be ZERO is
     ack/nack lines: pre-024 firmware beaconed those at 20 Hz.
  B  send HELLO. Identity authority per CLAUDE.md -- proves which board
     answered and that the firmware is alive at all.
  C  quiet again 3.0 s after HELLO's own reply is consumed. Zero ack/nack.
  D  send 'TLM POSE #1' (sequenced -- v6 drops unsequenced lines) and
     confirm telemetry frames AND the piggybacked reliability line both
     resume, i.e. ticket 001 did not break the TLM path.
"""
import sys
import time

import serial

PORT = '/dev/cu.usbmodem2121102'
BAUD = 115200


def listen(p, seconds, label):
    """Collect complete lines for `seconds`, sending nothing."""
    end = time.time() + seconds
    buf = b''
    lines = []
    while time.time() < end:
        chunk = p.read(4096)
        if chunk:
            buf += chunk
            while b'\n' in buf:
                raw, buf = buf.split(b'\n', 1)
                t = raw.decode('ascii', errors='replace').strip()
                if t:
                    lines.append((round(time.time(), 3), t))
    print(f'--- {label}: {len(lines)} line(s) over {seconds}s ---')
    for ts, t in lines:
        print(f'    {ts}  {t}')
    if not lines:
        print('    (silence -- zero bytes)')
    return [t for _, t in lines]


def rel(lines):
    return [t for t in lines if t.startswith('ack ') or t.startswith('nack ')]


p = serial.Serial(PORT, BAUD, timeout=0.1)
print(f'opened {PORT} @ {BAUD} (this resets the target program)')

# --- Phase A: fresh boot, nothing sent -----------------------------------
a = listen(p, 3.0, 'A  post-open, NOTHING sent')
a_rel = rel(a)

# --- Phase B: identity ---------------------------------------------------
p.reset_input_buffer()
p.write(b'HELLO\n')
p.flush()
b = listen(p, 1.5, 'B  after HELLO (reply expected)')
banner = [t for t in b if t.startswith('device ')]

# --- Phase C: quiet after a request has been served ----------------------
c = listen(p, 3.0, 'C  quiet window after HELLO')
c_rel = rel(c)

# --- Phase D: telemetry piggyback regression -----------------------------
p.reset_input_buffer()
p.write(b'TLM POSE #1\n')
p.flush()
d = listen(p, 2.0, 'D  after TLM POSE #1')
d_t = [t for t in d if t.startswith('t ') or t.startswith('thdr ')]
d_rel = rel(d)

p.write(b'TLM OFF #2\n')
p.flush()
time.sleep(0.3)
p.close()

print()
print('=' * 62)
print('RESULT')
print('=' * 62)
print(f'A  ack/nack while idle post-boot : {len(a_rel)}   (want 0)')
print(f'B  banner seen                   : {banner[0] if banner else "NONE"}')
print(f'C  ack/nack in quiet window      : {len(c_rel)}   (want 0)')
print(f'D  telemetry frames after TLM    : {len(d_t)}   (want > 0)')
print(f'D  piggybacked reliability lines : {len(d_rel)}   (want > 0)')

ok = (not a_rel) and banner and (not c_rel) and d_t and d_rel
print()
print('VERDICT:', 'PASS' if ok else 'FAIL')
sys.exit(0 if ok else 1)
